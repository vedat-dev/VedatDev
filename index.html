const express = require('express'); const cors = require('cors'); const fetch = require('node-fetch'); const app = express(); const PORT = process.env.PORT || 3000; app.use(cors()); app.use(express.json()); // Helper for Roblox API Requests async function robloxFetch(url, options = {}) { const headers = { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36', 'Accept': 'application/json', ...options.headers }; const res = await fetch(url, { ...options, headers }); if (!res.ok) { const errorText = await res.text(); const err = new Error(`Roblox API HTTP ${res.status}: ${errorText}`); err.status = res.status; throw err; } return res.json(); } // Service Status Check Endpoint app.get('/api/status', async (req, res) => { try { await robloxFetch('https://users.roblox.com/v1/users/1'); res.json({ status: 'ONLINE', timestamp: new Date().toISOString() }); } catch (e) { res.status(503).json({ status: 'UNAVAILABLE', message: 'Roblox API reachability check failed' }); } }); // Profile Scanner Intelligence API Endpoint app.get('/api/profile', async (req, res) => { const { username } = req.query; if (!username || typeof username !== 'string' || !username.trim()) { return res.status(400).json({ error: 'USERNAME_REQUIRED', message: 'Please provide a valid Roblox username.' }); } try { // 1. Resolve Username to User ID const resolveRes = await robloxFetch('https://users.roblox.com/v1/usernames/users', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ usernames: [username.trim()], excludeBannedUsers: false }) }); if (!resolveRes.data || resolveRes.data.length === 0) { return res.status(404).json({ error: 'USER_NOT_FOUND', message: `No Roblox account found with username "${username}".` }); } const userBasic = resolveRes.data[0]; const userId = userBasic.id; // 2. Fetch User Details & Thumbnails in Parallel const [userDetails, avatarData, headshotData] = await Promise.allSettled([ robloxFetch(`https://users.roblox.com/v1/users/${userId}`), robloxFetch(`https://thumbnails.roblox.com/v1/users/avatar?userIds=${userId}&size=720x720&format=Png&isCircular=false`), robloxFetch(`https://thumbnails.roblox.com/v1/users/avatar-headshot?userIds=${userId}&size=420x420&format=Png&isCircular=false`) ]); if (userDetails.status !== 'fulfilled') { return res.status(500).json({ error: 'PROFILE_FETCH_FAILED', message: 'Failed to retrieve user profile details from Roblox.' }); } const profile = userDetails.value; const avatarUrl = avatarData.status === 'fulfilled' && avatarData.value.data?.[0]?.imageUrl ? avatarData.value.data[0].imageUrl : null; const headshotUrl = headshotData.status === 'fulfilled' && headshotData.value.data?.[0]?.imageUrl ? headshotData.value.data[0].imageUrl : null; // 3. Fetch User Badges let badges = []; try { const badgeRes = await robloxFetch(`https://badges.roblox.com/v1/users/${userId}/badges?limit=25&sortOrder=Desc`); if (badgeRes && badgeRes.data) { badges = badgeRes.data.map(b => ({ id: b.id, name: b.name, description: b.description || 'No description provided.', awardDate: b.awardDate || null, iconUrl: b.iconImageId ? `https://assetgame.roblox.com/Game/Tools/Avatar.ashx?aid=${b.iconImageId}` : null })); } } catch (e) { badges = []; } // 4. Fetch Collectibles / Limited Items let inventory = { status: 'AVAILABLE', items: [], totalCount: 0, totalRAP: 0, highestRAP: 0, averageRAP: 0, limitedCount: 0, limitedUCount: 0 }; try { const inventoryRes = await robloxFetch(`https://inventory.roblox.com/v1/users/${userId}/assets/collectibles?limit=100&sortOrder=Desc`); if (inventoryRes && inventoryRes.data) { const rawItems = inventoryRes.data; const processedItems = rawItems.map(item => { const isLimitedU = item.assetStock !== undefined && item.assetStock !== null; const rap = item.recentAveragePrice || 0; const value = item.originalPrice || rap; return { assetId: item.assetId, userAssetId: item.userAssetId, name: item.name, rap: rap, value: value, serialNumber: item.serialNumber || null, assetStock: item.assetStock || null, isLimited: true, isLimitedU: isLimitedU, assetType: item.assetType || 'Accessory', thumbnailUrl: `https://assetdelivery.roblox.com/v1/asset/?id=${item.assetId}` }; }); const totalRAP = processedItems.reduce((acc, curr) => acc + curr.rap, 0); const highestRAP = processedItems.length > 0 ? Math.max(...processedItems.map(i => i.rap)) : 0; const limitedCount = processedItems.filter(i => !i.isLimitedU).length; const limitedUCount = processedItems.filter(i => i.isLimitedU).length; inventory.items = processedItems; inventory.totalCount = processedItems.length; inventory.totalRAP = totalRAP; inventory.highestRAP = highestRAP; inventory.averageRAP = processedItems.length > 0 ? Math.round(totalRAP / processedItems.length) : 0; inventory.limitedCount = limitedCount; inventory.limitedUCount = limitedUCount; } } catch (invError) { if (invError.status === 403) { inventory.status = 'PRIVATE'; } else { inventory.status = 'UNAVAILABLE'; } } res.json({ user: { id: profile.id, username: profile.name, displayName: profile.displayName, created: profile.created, isBanned: profile.isBanned || false, description: profile.description || '', externalUrl: `https://www.roblox.com/users/${profile.id}/profile`, avatarUrl: avatarUrl, headshotUrl: headshotUrl }, inventory: inventory, badges: badges, scannedAt: new Date().toISOString() }); } catch (error) { console.error('Scan Error:', error.message); if (error.status === 429) { return res.status(429).json({ error: 'RATE_LIMITED', message: 'Roblox API rate limit reached. Please wait a moment before trying again.' }); } res.status(500).json({ error: 'SERVER_ERROR', message: 'An internal error occurred while analyzing the profile.' }); } }); // Serve Full Frontend Client HTML Directly from Root app.get('/', (req, res) => { res.send(` SHADOWZ // Roblox Profile Intelligence
